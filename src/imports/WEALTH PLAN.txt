const fmt = n => {
  if (n >= 1e7) return '₹' + (n/1e7).toFixed(2) + ' Cr';
  if (n >= 1e5) return '₹' + (n/1e5).toFixed(2) + ' L';
  return '₹' + Math.round(n).toLocaleString('en-IN');
};
const fmtPct = n => n.toFixed(1) + '%';

  const v = parseFloat(document.getElementById(rid).value);
  document.getElementById(fid).style.width = ((v-mn)/(mx-mn)*100) + '%';
};

// ════════════════════════════════════════════════════
//  WEALTH CALCULATOR
// ════════════════════════════════════════════════════
function calculate() {
  const P   = parseFloat(document.getElementById('principal').value)  || 0;
  const m   = parseFloat(document.getElementById('monthly').value)    || 0;
  const g   = parseFloat(document.getElementById('growth').value)    / 100;
  const inf = parseFloat(document.getElementById('inflation').value) / 100;
  const Y   = parseInt(document.getElementById('years').value);

  currentInf = inf * 100;
  currentYrs = Y;

  const labels=[], nomData=[], realData=[], depData=[];
  let nom=P, real=P;
  for (let yr=0; yr<=Y; yr++) {
    labels.push(yr===0 ? 'Now' : 'Yr '+yr);
    nomData.push(Math.round(nom));
    realData.push(Math.round(real));
    depData.push(Math.round(P + m*12*yr));
    nom  = nom*(1+g) + m*12;
    real = real*(1+g)/(1+inf) + m*12/(1+inf);
  }

  const fn=nomData[Y], fr=realData[Y], dep=P+m*12*Y;
  const prof=fn-dep, roi=dep>0?(prof/dep)*100:0;
  const rg=((1+g)/(1+inf)-1)*100;

  document.getElementById('finalBalance').textContent  = fmt(fn);
  document.getElementById('realPower').textContent     = fmt(fr);
  document.getElementById('erosion').textContent       = fmt(fn-fr);
  document.getElementById('totalInvested').textContent = fmt(dep);
  document.getElementById('totalProfit').textContent   = fmt(prof);
  document.getElementById('roi').textContent           = fmtPct(roi);
  document.getElementById('realGrowth').textContent    = fmtPct(rg);

  const ctx = document.getElementById('wealthChart').getContext('2d');
  if (wealthChart) wealthChart.destroy();
  wealthChart = new Chart(ctx, {
    type:'line',
    data:{
      labels,
      datasets:[
        {label:'Nominal Wealth', data:nomData, borderColor:'#1A4A2A', backgroundColor:'rgba(26,74,42,0.08)', borderWidth:2.5, pointRadius:0, pointHoverRadius:5, fill:true, tension:0.4},
        {label:'Real Value',     data:realData, borderColor:'#C9A55A', backgroundColor:'rgba(201,165,90,0.05)', borderWidth:2, pointRadius:0, pointHoverRadius:5, fill:true, tension:0.4, borderDash:[6,3]},
        {label:'Deposited',      data:depData, borderColor:'rgba(17,24,39,0.22)', backgroundColor:'transparent', borderWidth:1.5, pointRadius:0, fill:false, tension:0, borderDash:[3,4]},
      ]
    },
    options:{
      responsive:true, maintainAspectRatio:false,
      interaction:{mode:'index',intersect:false},
      plugins:{
        legend:{display:false},
        tooltip:{
          backgroundColor:'#fff', titleColor:'#111827', bodyColor:'#6b7280',
          borderColor:'rgba(0,0,0,0.1)', borderWidth:1, padding:12,
          titleFont:{family:'Playfair Display',size:13,weight:'700'},
          bodyFont:{family:'Nunito Sans',size:12},
          callbacks:{label:c=>' '+c.dataset.label+': '+fmt(c.raw)}
        }
      },
      scales:{
        x:{grid:{color:'rgba(0,0,0,0.04)'},ticks:{font:{family:'Nunito Sans',size:10},color:'#9ca3af',maxTicksLimit:10}},
        y:{grid:{color:'rgba(0,0,0,0.04)'},ticks:{font:{family:'Nunito Sans',size:10},color:'#9ca3af',callback:v=>fmt(v)}}
      }
    }
  });

  document.getElementById('stYrs').textContent  = Y;
  document.getElementById('stRate').textContent = (inf*100).toFixed(1)+'%';
  renderInfResult();
}

function populateInfItems() {
  const cat = document.getElementById('catPick').value;
  const items = ALL_ITEMS.filter(i => i.cat === cat);
  document.getElementById('itemPick').innerHTML = items.map((it, i) =>
    `<option value="${i}">${it.icon} ${it.name} — ₹${it.price.toLocaleString('en-IN')}</option>`
  ).join('');
  renderInfResult();
}

function renderInfResult() {
  const cat = document.getElementById('catPick').value;
  const items = ALL_ITEMS.filter(i => i.cat === cat);
  const idx = parseInt(document.getElementById('itemPick').value) || 0;
  const it = items[idx];
  if (!it) return;
  const inf = currentInf/100, yrs = currentYrs;
  const future = Math.round(it.price * Math.pow(1+inf, yrs));
  const mult = (future/it.price).toFixed(1);

  document.getElementById('infResIcon').textContent  = it.icon;
  document.getElementById('infResName').textContent  = it.name;
  document.getElementById('infResNow').textContent   = '₹' + it.price.toLocaleString('en-IN');
  document.getElementById('infResFuture').textContent= '₹' + future.toLocaleString('en-IN');
  document.getElementById('infResMult').textContent  = mult + '× costlier';
}

function initHistorical() {
  // Build pills
  document.getElementById('histPills').innerHTML = HIST_DATA.map(d => {
    const mult = (d.prices[d.prices.length-1] / d.prices[0]).toFixed(1);
    return `<div class="hist-pill ${d.id===activeHist?'active':''}" onclick="selectHist('${d.id}')">
      <div class="hp-ico">${d.icon}</div>
      <div>
        <div class="hp-nm">${d.name}</div>
        <div class="hp-change">${mult}× since 2000</div>
      </div>
    </div>`;
  }).join('');

  renderHistChart(activeHist);
  renderThenNow();
}

function selectHist(id) {
  activeHist = id;
  document.querySelectorAll('.hist-pill').forEach(p => {
    p.classList.toggle('active', p.getAttribute('onclick') === `selectHist('${id}')`);
  });
  renderHistChart(id);
}

function renderHistChart(id) {
  const d = HIST_DATA.find(x=>x.id===id);
  document.getElementById('histChartTitle').textContent = d.name + ' — Price History (India)';
  const ctx = document.getElementById('histChart').getContext('2d');
  if (histChart) histChart.destroy();
  histChart = new Chart(ctx, {
    type:'line',
    data:{
      labels: d.years.map(String),
      datasets:[{
        label: d.unit,
        data: d.prices,
        borderColor: d.color,
        backgroundColor: d.color+'1a',
        borderWidth: 2.5,
        pointRadius: 5,
        pointHoverRadius: 7,
        pointBackgroundColor: d.color,
        pointBorderColor: '#fff',
        pointBorderWidth: 2,
        fill: true,
        tension: 0.35,
      }]
    },
    options:{
      responsive:true, maintainAspectRatio:false,
      plugins:{
        legend:{display:false},
        tooltip:{
          backgroundColor:'#fff', titleColor:'#111827', bodyColor:'#6b7280',
          borderColor:'rgba(0,0,0,0.1)', borderWidth:1, padding:12,
          titleFont:{family:'Playfair Display',size:13,weight:'700'},
          bodyFont:{family:'Nunito Sans',size:12},
          callbacks:{label:c=>' ₹'+c.raw.toLocaleString('en-IN')}
        }
      },
      scales:{
        x:{grid:{color:'rgba(0,0,0,0.04)'},ticks:{font:{family:'Nunito Sans',size:11},color:'#9ca3af'}},
        y:{
          grid:{color:'rgba(0,0,0,0.04)'},
          ticks:{font:{family:'Nunito Sans',size:11},color:'#9ca3af',callback:v=>'₹'+(v>=1000?(v/1000).toFixed(0)+'k':v)}
        }
      }
    }
  });
}

function renderThenNow() {
  document.getElementById('tnGrid').innerHTML = HIST_DATA.map(d => {
    const old = d.prices[0], now = d.prices[d.prices.length-1];
    const mult = (now/old);
    const yrs = d.years[d.years.length-1] - d.years[0];
    const cagr = ((Math.pow(mult, 1/yrs)-1)*100).toFixed(1);
    const barW = Math.min(100, (mult/20)*100);
    return `<div class="tn-card">
      <div class="tn-head"><span class="tn-ico">${d.icon}</span><span class="tn-nm">${d.name}</span></div>
      <div class="tn-prices">
        <div class="tn-col"><div class="tn-yr">Year 2000</div><div class="tn-old">₹${old.toLocaleString('en-IN')}</div></div>
        <div class="tn-arr">→</div>
        <div class="tn-col r"><div class="tn-yr" style="color:var(--red);opacity:0.7">Today 2026</div><div class="tn-new">₹${now.toLocaleString('en-IN')}</div></div>
      </div>
      <div class="tn-bar-wrap"><div class="tn-bar" style="width:${barW}%"></div></div>
      <div class="tn-cagr">${mult.toFixed(1)}× costlier &nbsp;·&nbsp; CAGR <span>${cagr}%</span>/yr</div>
    </div>`;
  }).join('');
}
